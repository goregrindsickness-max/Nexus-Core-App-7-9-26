  const initYoutubePlayer = (videoId: string) => {
    const win = window as any;
    if (!win.YT || !win.YT.Player) {
      setTimeout(() => initYoutubePlayer(videoId), 500);
      return;
    }

    if (ytPlayerRef.current) {
      try {
        if (typeof ytPlayerRef.current.loadVideoById === 'function') {
          if (sceneRadioPlaying) {
            ytPlayerRef.current.loadVideoById(videoId);
          } else {
            ytPlayerRef.current.cueVideoById(videoId);
          }
          return;
        }
      } catch (err) {
        console.warn("[RADIO PLAYER] Reusing player failed, recreating...", err);
      }
    }

    try {
      ytPlayerRef.current = new win.YT.Player('youtube-radio-player', {
        height: '100%',
        width: '100%',
        videoId: videoId,
        playerVars: {
          autoplay: sceneRadioPlaying ? 1 : 0,
          controls: 0,
          disablekb: 1,
          fs: 0,
          rel: 0,
          modestbranding: 1,
          iv_load_policy: 3,
        },
        events: {
          onReady: () => {
            console.log("[RADIO PLAYER] YT Player is Ready.");
            if (sceneRadioPlaying && ytPlayerRef.current && typeof ytPlayerRef.current.playVideo === 'function') {
              ytPlayerRef.current.playVideo();
            }
          },
          onStateChange: (event: any) => {
            if (event.data === 1) {
              setSceneRadioPlaying(true);
              setRadioPlayerError(null);
            } else if (event.data === 2) {
              setSceneRadioPlaying(false);
            } else if (event.data === 0) {
              handleNextSong();
            }
          },
          onError: (event: any) => {
            console.warn("[RADIO PLAYER] Error code:", event.data);
            let errorMsg = "Playback error. Skipping...";
            if (event.data === 101 || event.data === 150) {
              errorMsg = "Age-restricted or embedding blocked by owner. Auto-skipping...";
            } else if (event.data === 100) {
              errorMsg = "Video not found. Skipping...";
            }
            setRadioPlayerError(errorMsg);
            
            setTimeout(() => {
              handleNextSong();
            }, 2500);
          }
        }
      });
    } catch (e) {
      console.error("[RADIO PLAYER] Error initializing YT Player:", e);
    }
  };

  const togglePlayPause = () => {

    try {

        ytPlayerRef.current.pauseVideo();
        setSceneRadioPlaying(false);
      } else {
        ytPlayerRef.current.playVideo();
        setSceneRadioPlaying(true);
      }
    } catch (err) {
      console.error("[RADIO PLAYER] Error calling play/pause on YT player:", err);
    }
